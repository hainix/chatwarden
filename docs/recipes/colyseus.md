# Recipe: Colyseus

This one is written by hand. Init does not write it, because a generated file that imports Colyseus would have to be tested against a real Colyseus on every release, and that dependency is large and moves fast.

**You write:** `onAuth` in your room, where you decide who the player is. Everything else is the `attach(room, kit)` helper below, which you copy into your project.

**Init writes:** nothing for Colyseus itself. It can write the kit (`kit.ts`) that the helper takes: the helper needs the kit's `chat` room broadcaster, so ask for the chat box.

```sh
npx chatwarden init --framework none --chat box
```

## Which Colyseus

The helper follows Colyseus 0.16: `Room.onAuth`, `onJoin`, `onLeave` and `onMessage`, and `Client.send` and `Client.leave`. Colyseus 0.15 and 0.16 differ in their room API, so check the file against the version you run.

It was checked against fakes of `Room` and `Client` only. It has not been run in a real Colyseus. The fakes are small structural types in `examples/recipes/colyseus-types.d.ts`. In your project, import the real types and delete the fakes.

## What it does

- `onAuth` returns who the player is and the name other players see. Take the identity from your own account or session, never from a value the client chose.
- `onJoin` opens a chat session for the room. Every frame (the hello, lines, hides, notices, the goodbye) goes to the client as `chat:frame`.
- The client sends `chat:send`, `chat:mute` and `chat:report`. Each answer comes back as `chat:result`.
- `onLeave` closes the stream.

Every line still goes through the kit's chat check, with the same limits as the HTTP routes.

```ts
// Chatwarden for a Colyseus room: attach(room, kit) puts the moderated chat on the room's messages.
//
// This follows the shape of Colyseus 0.16 (Room.onAuth, onJoin, onLeave, onMessage; Client.send and Client.leave). Colyseus 0.15 and 0.16 differ in
// their room API, so check this file against the version you run. It is not generated by `chatwarden init`, and it has been checked against fakes of
// Room and Client only (colyseus-types.d.ts in this folder's parent, and examples/recipes/colyseus.test.ts): it has not been run in a real Colyseus.
//
// What your game adds: onAuth is where you identify the player. Return a ChatAuth ({ who, member }) from it: who is the player's identity (your own
// account id or session, never a value the client chose) and member.name is the name other players see. Colyseus hands what onAuth returned to onJoin
// as its third argument, and that is where attach() opens the chat session.
//
// The messages a client sends, and what it gets back:
//   'chat:send'   { text }                  answers 'chat:result' { type: 'send', result }
//   'chat:mute'   { lineId } or { ref }, on answers 'chat:result' { type: 'mute', result }
//   'chat:report' { lineId, reason?, note? } answers 'chat:result' { type: 'report', outcome }
// and the room sends every frame (the hello, lines, hides, notices, bye) as 'chat:frame'.
import type { RawWho } from '@chatwarden/core';
import type { ChatMember, ChatSession, ChatStream, RoomBroadcaster } from '@chatwarden/http/chat';
import type { Client, Room } from '../../colyseus-types.js';

/** What your onAuth returns for the chat: who the player is, and the name to show. */
export interface ChatAuth {
  who: RawWho;
  member: ChatMember;
}

// The same limits the HTTP chat routes use.
const TEXT_MAX = 2000;
const NOTE_MAX = 500;
const REASON_MAX = 64;
const ID = /^[A-Za-z0-9_-]{1,64}$/;

const isObject = (value: unknown): value is Record<string, unknown> => typeof value === 'object' && value !== null && !Array.isArray(value);
const idOf = (value: unknown): value is string => typeof value === 'string' && ID.test(value);

/**
 * Put the chat on `room`: register the messages above, open a session when a client joins and close it when they leave. `kit` is your Chatwarden kit
 * (its `chat` is the room broadcaster). `roomId` is the chat room's id (default: the Colyseus room's id); the kit's join hook is not used here, since
 * onAuth already decided who may be in this room.
 */
export function attach(room: Room, kit: { chat: RoomBroadcaster }, roomId: string = room.roomId): void {
  const live = new Map<string, { session: ChatSession; stream: ChatStream }>();
  const result = (client: Client, body: Record<string, unknown>): void => client.send('chat:result', body);

  room.onMessage('chat:send', (client, input) => {
    const entry = live.get(client.sessionId);
    if (entry === undefined) return;
    const text = isObject(input) && typeof input.text === 'string' ? input.text : '';
    result(client, { type: 'send', result: text.length > TEXT_MAX ? { ok: false, refused: { reason: 'length' } } : entry.session.send({ text }) });
  });

  room.onMessage('chat:mute', (client, input) => {
    const entry = live.get(client.sessionId);
    if (entry === undefined) return;
    const byLine = isObject(input) && idOf(input.lineId);
    const byRef = isObject(input) && idOf(input.ref);
    if (!isObject(input) || typeof input.on !== 'boolean' || byLine === byRef) return result(client, { type: 'mute', result: { ok: false, hiddenIds: [] } });
    const target = byLine ? { lineId: input.lineId as string } : { ref: input.ref as string };
    result(client, { type: 'mute', result: entry.session.muteResult({ ...target, on: input.on }) });
  });

  room.onMessage('chat:report', (client, input) => {
    const entry = live.get(client.sessionId);
    if (entry === undefined) return;
    if (!isObject(input) || !idOf(input.lineId)) return result(client, { type: 'report', outcome: { ok: false, refused: 'not_found' } });
    const reason = typeof input.reason === 'string' && input.reason.length <= REASON_MAX ? input.reason : undefined;
    const note = typeof input.note === 'string' && input.note.length <= NOTE_MAX ? input.note : undefined;
    entry.session
      .report({ lineId: input.lineId, ...(reason === undefined ? {} : { reason }), ...(note === undefined ? {} : { note }) })
      .then(
        (outcome) => result(client, { type: 'report', outcome }),
        () => result(client, { type: 'report', outcome: { ok: false, refused: 'not_found' } }),
      );
  });

  // Your own onJoin and onLeave run first; the chat is added after them.
  const gameJoin = room.onJoin;
  room.onJoin = async (client, options, auth) => {
    await gameJoin?.call(room, client, options, auth);
    const chat = auth as ChatAuth | undefined;
    if (chat === undefined || chat.who === undefined || chat.member === undefined) return void client.leave(4001, 'unauthorized');
    const session = kit.chat.session(roomId, chat.who, chat.member);
    if ('refused' in session) return void client.leave(session.refused.reason === 'unauthorized' ? 4001 : 4003, session.refused.reason);
    const stream = session.open({
      write: (frame) => {
        client.send('chat:frame', frame);
        return true;
      },
      end: () => client.leave(4000, 'closed'),
    });
    if ('refused' in stream) return void client.leave(4003, stream.refused.reason);
    live.set(client.sessionId, { session, stream });
  };

  const gameLeave = room.onLeave;
  room.onLeave = async (client, consented) => {
    await gameLeave?.call(room, client, consented);
    live.get(client.sessionId)?.stream.close();
    live.delete(client.sessionId);
  };
}
```
